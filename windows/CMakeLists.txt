cmake_minimum_required(VERSION 3.21)
project(SolasiModels LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
option(SOLASI_CPU_AVX2 "Build the guarded AVX2/FMA/F16C CPU worker variant" OFF)
if(WIN32)
    foreach(feature SSE42 AVX AVX2 FMA F16C BMI2)
        set(GGML_${feature} ${SOLASI_CPU_AVX2} CACHE BOOL "" FORCE)
    endforeach()
    foreach(feature AVX512 AVX_VNNI AVX512_VBMI AVX512_VNNI AVX512_BF16)
        set(GGML_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
endif()
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)
get_property(SOLASI_VENDOR_LICENSES GLOBAL PROPERTY LICENSE_TEXT)
file(WRITE "${CMAKE_BINARY_DIR}/vendor-licenses.txt" "${SOLASI_VENDOR_LICENSES}")
add_executable(ModelWorker ModelWorker.cpp)
target_link_libraries(ModelWorker PRIVATE llama-common mtmd)
target_include_directories(ModelWorker PRIVATE "${LLAMA_SOURCE}/common" "${LLAMA_SOURCE}/tools/mtmd")
add_executable(EmbeddingWorker ../macos/EmbeddingWorker.cpp)
target_link_libraries(EmbeddingWorker PRIVATE llama)
add_executable(WindowsCpu WindowsCpu.cpp)
if(MSVC)
    target_compile_options(ModelWorker PRIVATE /utf-8)
    target_compile_options(EmbeddingWorker PRIVATE /utf-8)
endif()
